# 示例

以下片段均可在仓库 `demo/` 中找到完整可运行版本（仓库根执行 `npm run dev:demo`，端口 9303）。

## 一、设计器接入 + 保存 + 免保存预览

```vue
<template>
  <PrintDesigner
    ref="designerRef"
    :initial-template="templateData"
    :fields="fields"
    :is-edit="true"
    :show-help="true"
    @preview="onPreview"
    @save="onSave"
  />

  <Teleport to="body">
    <div v-if="previewVisible" class="preview-mask" @click.self="previewVisible = false">
      <div class="preview-panel">
        <button type="button" @click="htmlPreviewRef?.print()">打印</button>
        <span v-if="previewPages > 0">共 {{ previewPages }} 页</span>
        <PrintHtmlPreview
          ref="htmlPreviewRef"
          :template-json="previewJson"
          :print-data="printData"
          :base-url="baseUrl"
          @rendered="(n) => (previewPages = n)"
        />
      </div>
    </div>
  </Teleport>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import {
  PrintDesigner,
  PrintHtmlPreview,
  createDefaultTemplate,
} from '@worm-vue3-print/canvas'
import type { PrintBusinessField, TemplateData } from '@worm-vue3-print/canvas'
import '@worm-vue3-print/canvas/native-controls.css'

const designerRef = ref<InstanceType<typeof PrintDesigner> | null>(null)
const htmlPreviewRef = ref<InstanceType<typeof PrintHtmlPreview> | null>(null)

const templateData = ref<TemplateData>(createDefaultTemplate())
const fields = ref<PrintBusinessField[]>([
  // 分组按 fieldKey 前缀结构自动判定，fieldType 与叶子字段一致即可（demo 用 'string'）
  { fieldKey: 'supplier', fieldLabel: '供应商', fieldType: 'string', sortOrder: 1 },
  { fieldKey: 'supplier.name', fieldLabel: '名称', fieldType: 'text', sortOrder: 2 },
  { fieldKey: 'goods', fieldLabel: '商品明细', fieldType: 'list', sortOrder: 3 },
  { fieldKey: 'goods.spec', fieldLabel: '规格', fieldType: 'text', sortOrder: 4 },
  { fieldKey: 'goods.qty', fieldLabel: '数量', fieldType: 'number', sortOrder: 5 },
])

const printData = ref({
  supplier: { name: 'ACME 公司' },
  goods: [
    { spec: '红色 / L', qty: 2 },
    { spec: '蓝色 / M', qty: 5 },
  ],
})
const baseUrl = ''

const previewVisible = ref(false)
const previewPages = ref(0)
const previewJson = ref<Record<string, any> | null>(null)

function onPreview() {
  const json = designerRef.value?.getTemplateJson?.()
  if (!json) return
  previewJson.value = json as unknown as Record<string, any>
  previewPages.value = 0
  previewVisible.value = true
}

function onSave(json: string) {
  // 由宿主持久化；demo 里直接下载 JSON 文件
  const blob = new Blob([JSON.stringify(JSON.parse(json), null, 2)], {
    type: 'application/json',
  })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `template-${Date.now()}.json`
  a.click()
  URL.revokeObjectURL(url)
}
</script>
```

## 二、程序化调用渲染管线（不用设计器组件）

```ts
import { renderHtmlPages, browserCodeRenderer } from '@worm-vue3-print/core/browser'
import type { PrintTemplateData } from '@worm-vue3-print/core'

const { html, pageCount, paperMm, continuous } = await renderHtmlPages(
  templateJson as PrintTemplateData,
  printData,
  baseUrl,
  browserCodeRenderer,
)

// 写入新窗口打印
const win = window.open('', '_blank')
win?.document.open()
win?.document.write(html)
win?.document.close()
win?.focus()
win?.print()
```

## 三、请求服务端 PDF（同源代理 + 新标签打开）

```ts
// 密钥由宿主后端 / dev 代理在 X-Render-Key 中注入，不出现在前端
async function requestServerPdf(
  templateJson: Record<string, unknown>,
  printData: Record<string, unknown>,
  baseUrl: string,
): Promise<Blob> {
  const res = await fetch('/render-api/render/pdf', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ templateJson, printData, baseUrl }),
  })
  if (!res.ok) {
    const errBody = await res.json().catch(() => null)
    throw new Error(errBody?.message ?? `HTTP ${res.status}`)
  }
  return res.blob()
}

function openPdfBlob(pdf: Blob, filename: string) {
  const url = URL.createObjectURL(pdf)
  const win = window.open(url, '_blank')
  if (!win) {
    // 被弹窗拦截时降级下载
    const a = document.createElement('a')
    a.href = url
    a.download = filename
    a.click()
  }
  setTimeout(() => URL.revokeObjectURL(url), 60_000)
}

const pdf = await requestServerPdf(templateJson, printData, baseUrl)
openPdfBlob(pdf, `print-${Date.now()}.pdf`)
```

健康检查：`GET /render-api/health` →
`{ status, activeRenders, maxConcurrent, queueLength }`。

## 四、静默打印（浏览器预渲染 + printHtml 直送）

```ts
import { PrintClient, WormPrintError } from '@worm-vue3-print/core/client'
import { renderHtmlPages, browserCodeRenderer } from '@worm-vue3-print/core/browser'

const client = new PrintClient({ timeoutMs: 20_000 })
client.onStatusChange((s) => {
  // 'disconnected' | 'connecting' | 'connected'
  console.log('客户端状态', s)
})

await client.connect()
const printers = await client.listPrinters()

// 页面内两遍渲染
const rendered = await renderHtmlPages(
  templateJson as any,
  printData,
  baseUrl,
  browserCodeRenderer,
)

try {
  const { jobId } = await client.printHtml(
    rendered,
    { printerName: printers[0]?.name, copies: 1 },
    '采购收货单',
  )
  console.log('已提交，作业号', jobId)
} catch (e) {
  if (e instanceof WormPrintError) {
    // e.code: BUSY / CLIENT_NOT_RUNNING / PRINTER_OFFLINE ...
    console.error(e.code, e.message)
  }
}
```

兼容链路（客户端内渲染）：

```ts
await client.print(templateJson, printData, {
  printerName: '热敏-80',
  copies: 1,
  // 长度单位 μm：1mm = 1000μm；连续纸高度逃生门示例 180mm
  paperSize: { width: 80_000, height: 180_000 },
})
```

开启安全配对后先执行一次 `client.pair(token)`，详见《静默打印》。

## 五、表达式引擎与自定义函数

```ts
import { TemplateEngine } from '@worm-vue3-print/core'

const engine = new TemplateEngine({
  functions: {
    MONEY: (x: number) => x.toFixed(2),
  },
})

engine.render('合计：{MONEY(qty * price)} 元', { qty: 3, price: 12.5 })
// '合计：37.50 元'

engine.registerFunction('DOUBLE', (x: number) => x * 2)
engine.evaluate('DOUBLE(amount)', { amount: 21 }) // 42
```

模板经渲染管线时还可直接使用内置的 `MONEY/DATE/UPPER/IF/CONCAT/IFEMPTY/LEN`
与数值运算 `ADD/SUB/MUL/DIV`、修约 `ROUND/ROUNDUP(CEIL)/ROUNDDOWN(FLOOR)/ROUNDBANK`、
聚合 `SUM/AVG/COUNT/MIN/MAX`，以及系统变量 `{printDate}`、`{pageIndex}`、`{totalPages}`。
四则运算也可直接写运算符（`{qty * price}`、`{ROUND(qty * price, 2)}`）。

## 六、宿主能力注入：截图与上传

```ts
import type { RequestScreenshotFn, UploadImageFn } from '@worm-vue3-print/canvas'

const requestScreenshot: RequestScreenshotFn = async ({ templateJson, printData }) => {
  const res = await fetch('/render-api/render/screenshot', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ templateJson, printData }),
  })
  return res.blob()
}

const uploadImage: UploadImageFn = async (file) => {
  const form = new FormData()
  form.append('file', file)
  const res = await fetch('/api/files/upload', { method: 'POST', body: form })
  const { url } = await res.json()
  return url
}
```

```vue
<PrintDesigner
  :initial-template="templateData"
  :fields="fields"
  :request-screenshot="requestScreenshot"
  :upload-image="uploadImage"
/>
```
